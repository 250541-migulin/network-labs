package network.lab1.client;

import java.io.*;
import java.net.Socket;

/**
 * TCP-клиент для ЛР1.
 * Подключается к серверу и отправляет команды.
 */
public class TcpClient {
    private final String host;
    private final int port;

    public TcpClient(String host, int port) {
        this.host = host;
        this.port = port;
    }

    /**
     * Запуск клиента: читаем команды из консоли, отправляем на сервер, выводим ответы.
     */
    public void start() {
        try (Socket socket = new Socket(host, port);
             BufferedReader in = new BufferedReader(new InputStreamReader(socket.getInputStream()));
             PrintWriter out = new PrintWriter(socket.getOutputStream(), true);
             BufferedReader console = new BufferedReader(new InputStreamReader(System.in))) {

            System.out.println("Подключено к " + host + ":" + port);
            System.out.println("Введите команду (ECHO <текст>, TIME, CLOSE):");

            String userInput;
            while ((userInput = console.readLine()) != null) {
                out.println(userInput);             // отправляем команду серверу
                String response = in.readLine();    // читаем ответ сервера
                System.out.println("Сервер: " + response);

                if ("CLOSE".equalsIgnoreCase(userInput)) {
                    break; // завершаем работу клиента
                }
            }

        } catch (IOException e) {
            e.printStackTrace(); // выводим ошибки сети
        }
    }
}

     * Выполнение команды.
     *
     * @param args аргументы команды
     * @param out поток для отправки ответа клиенту
     * @return true если после выполнения нужно закрыть соединение
     */
    boolean handle(String[] args, PrintWriter out);
}

package network.lab1.server;

/**
 * Спецификация команды: хранит допустимое количество аргументов
 * и обработчик, выполняющий команду.
 */
public record CommandSpec(
        int minArgs,
        int maxArgs,
        CommandHandler handler
) {

    /**
     * Проверка, что количество аргументов находится в допустимых пределах.
     */
    public boolean validate(String[] args) {
        return args.length >= minArgs && args.length <= maxArgs;
    }
}

package network.lab1.server;

import java.io.*;
import java.net.*;
import java.nio.file.Path;
import java.time.LocalDateTime;
import java.util.*;

/**
 * Последовательный TCP-сервер для ЛР1.
 * Поддерживает команды ECHO, TIME, CLOSE.
 */
public class TcpServer {
    private final int port;
    private final Map<String, CommandSpec> registry = new HashMap<>();

    public TcpServer(int port) {
        this.port = port;
        registerCommands(); // регистрируем команды при создании сервера
    }

    /**
     * Запуск сервера: слушаем порт и принимаем клиентов по очереди.
     */
    public void start() {
        try (ServerSocket serverSocket = new ServerSocket(port)) {
            System.out.println("Сервер слушает порт " + port);
            while (true) {
                Socket clientSocket = serverSocket.accept();
                System.out.println("Подключился клиент: " + clientSocket.getRemoteSocketAddress());
                handleClient(clientSocket);
            }
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    /**
     * Обработка одного клиента: читаем команды и отправляем ответы.
     */
    private void handleClient(Socket clientSocket) {
        try (BufferedReader in = new BufferedReader(new InputStreamReader(clientSocket.getInputStream()));
             PrintWriter out = new PrintWriter(clientSocket.getOutputStream(), true)) {

            String line;
            while ((line = in.readLine()) != null) {
                if (processCommand(line, out)) break; // завершение по CLOSE
            }
        } catch (IOException e) {
            e.printStackTrace();
        } finally {
            try { clientSocket.close(); } catch (IOException ignored) {}
            System.out.println("Клиент отключился");
        }
    }

    /**
     * Разбор и выполнение команды.
     * @return true если нужно закрыть соединение
     */
    private boolean processCommand(String line, PrintWriter out) {
        String[] parts = line.trim().split("\\s+");
        if (parts.length == 0) return false;

        String command = parts[0].toUpperCase();
        String[] args = Arrays.copyOfRange(parts, 1, parts.length);

        CommandSpec spec = registry.get(command);
        if (spec == null) {
            out.println("Неизвестная команда: " + command);
            return false;
        }
        if (!spec.validate(args)) {
            out.println("Ошибка: команда " + command +
                    " требует от " + spec.minArgs() + " до " + spec.maxArgs() + " аргументов");
            return false;
        }
        return spec.handler().handle(args, out);
    }

    /**
     * Регистрация стандартных команд.
     */
    private void registerCommands() {
        registry.put("ECHO", new CommandSpec(0, Integer.MAX_VALUE, (args, out) -> {
            out.println(String.join(" ", args));
            return false;
        }));

        registry.put("TIME", new CommandSpec(0, 0, (args, out) -> {
            out.println(LocalDateTime.now().toString());
            return false;
        }));

        registry.put("CLOSE", new CommandSpec(0, 0, (args, out) -> {
            out.println("Соединение закрыто");
            return true;
        }));

        registry.put("UPLOAD", new CommandSpec(1, 1, (args, ctx) -> {
            String filename = args[0];
            try {
                String lenLine = ctx.in().readLine();
                long length = Long.parseLong(lenLine.trim());

                Path target = Path.of(filename);
                TransferStats stats = receiveFile(ctx.rawIn(), target, length);

                ctx.out().println("Файл " + filename + " загружен. " +
                        "Размер: " + stats.bytes() + " байт, " +
                        "Скорость: " + String.format("%.2f", stats.bitrateKBps()) + " KB/s");
            } catch (IOException e) {
                ctx.out().println("Ошибка загрузки файла: " + e.getMessage());
            }
            return false;
        }));

        registry.put("DOWNLOAD", new CommandSpec(1, 1, (args, ctx) -> {
            String filename = args[0];
            Path source = Path.of(filename);

            if (!source.toFile().exists()) {
                ctx.out().println("Файл " + filename + " не найден");
                return false;
            }

            try {
                TransferStats stats = sendFile(ctx.rawOut(), source);
                ctx.out().println("Файл " + filename + " отправлен. " +
                        "Размер: " + stats.bytes() + " байт, " +
                        "Скорость: " + String.format("%.2f", stats.bitrateKBps()) + " KB/s");
            } catch (IOException e) {
                ctx.out().println("Ошибка скачивания файла: " + e.getMessage());
            }
            return false;
        }));

    }

    /**
     * Получает файл от клиента и сохраняет его на сервере.
     *
     * @param in входной поток клиента
     * @param target путь к файлу, куда сохраняем
     * @param length ожидаемый размер файла в байтах
     * @return количество байт и время передачи
     */
    private TransferStats receiveFile(InputStream in, Path target, long length) throws IOException {
        try (OutputStream fos = new FileOutputStream(target.toFile())) {
            long start = System.nanoTime();
            byte[] buffer = new byte[4096];
            long totalBytes = 0;
            int read;
            while (totalBytes < length && (read = in.read(buffer)) != -1) {
                fos.write(buffer, 0, read);
                totalBytes += read;
            }
            long end = System.nanoTime();
            return new TransferStats(totalBytes, end - start);
        }
    }

    /**
     * Отправляет файл клиенту.
     *
     * @param out выходной поток клиента
     * @param source путь к файлу на сервере
     * @return количество байт и время передачи
     */
    private TransferStats sendFile(OutputStream out, Path source) throws IOException {
        try (InputStream fis = new FileInputStream(source.toFile())) {
            long start = System.nanoTime();
            byte[] buffer = new byte[4096];
            long totalBytes = 0;
            int read;
            while ((read = fis.read(buffer)) != -1) {
                out.write(buffer, 0, read);
                totalBytes += read;
            }
            out.flush();
            long end = System.nanoTime();
            return new TransferStats(totalBytes, end - start);
        }
    }

}

package network.lab1.server;

/**
 * Вспомогательный класс для хранения статистики передачи.
 *
 * @param bytes количество переданных байт
 * @param nanos время передачи в наносекундах
 */
public record TransferStats(
        long bytes,
        long nanos
) {

    /**
     * Рассчитывает скорость передачи в килобайтах в секунду.
     *
     * @return скорость передачи (KB/s)
     */
    public double bitrateKBps() {
        double seconds = nanos / 1e9;
        return (bytes / 1024.0) / seconds;
    }
}
