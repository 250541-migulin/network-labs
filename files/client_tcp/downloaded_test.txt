public class ClientContext {
    private final SocketChannel channel;
    private final ByteBuffer readBuf = ByteBuffer.allocate(4096);
    private final Deque<ByteBuffer> outQueue = new ArrayDeque<>();

    // ��������� �������� �����
    private FileChannel fileChannel;
    private long filePos = 0;
    private long fileSize = 0;
    private boolean sendingFile = false;
    private boolean receivingFile = false;

    public ClientContext(SocketChannel channel) {
        this.channel = channel;

    /** ������ ������ �� ������� */
    public void read() throws IOException {
        int n = channel.read(readBuf);
        if (n == -1) throw new IOException("Client closed");
        readBuf.flip();

        // ����� ����� �����
        if (receivingFile && fileChannel != null) {
            while (readBuf.hasRemaining()) {
                fileChannel.write(readBuf);
                filePos += n;
            }
            readBuf.clear();
            if (filePos >= fileSize) {
                receivingFile = false;
                fileChannel.close();
                enqueue(Protocol.CTRL_DONE + Protocol.CRLF);
            }
            return;
        }

        // ����� ������
        while (readBuf.hasRemaining()) {
            byte b = readBuf.get();
            if (b == '\n') {
                String cmd = new String(readBuf.array(), 0, readBuf.position()).trim();
                handleCommand(cmd);
                readBuf.clear();
                break;
            }
        }
        readBuf.compact();
    }

    /** ��������� ������ */
    private void handleCommand(String cmd) {
        if (cmd.equalsIgnoreCase(Protocol.CMD_TIME)) {
            enqueue("Server time: " + LocalDateTime.now() + Protocol.CRLF);
        } else if (cmd.equalsIgnoreCase(Protocol.CMD_PING)) {
            enqueue("PONG" + Protocol.CRLF);
        } else if (cmd.startsWith(Protocol.CMD_DOWNLOAD)) {
            String[] parts = cmd.split(" ");
            if (parts.length < 2) {
                enqueue(Protocol.CTRL_ERROR + " usage DOWNLOAD <filename>" + Protocol.CRLF);
                return;
            }
            Path file = Path.of("files/server_tcp", parts[1]);
            try {
                fileChannel = FileChannel.open(file, StandardOpenOption.READ);
                fileSize = fileChannel.size();
                filePos = 0;
                sendingFile = true;
                enqueue(Protocol.CTRL_READY + " " + fileSize + Protocol.CRLF);
            } catch (IOException e) {
                enqueue(Protocol.CTRL_ERROR + " file not found" + Protocol.CRLF);
            }
        } else if (cmd.startsWith(Protocol.CMD_UPLOAD)) {
            String[] parts = cmd.split(" ");
            if (parts.length < 3) {
                enqueue(Protocol.CTRL_ERROR + " usage UPLOAD <filename> <size>" + Protocol.CRLF);
                return;
            }
            Path file = Path.of("files/server_tcp", parts[1]);
            try {
                fileSize = Long.parseLong(parts[2]);
                fileChannel = FileChannel.open(file,
                        StandardOpenOption.CREATE, StandardOpenOption.WRITE, StandardOpenOption.TRUNCATE_EXISTING);
                filePos = 0;
                receivingFile = true;
                enqueue(Protocol.CTRL_READY + Protocol.CRLF);
            } catch (IOException e) {
                enqueue(Protocol.CTRL_ERROR + " cannot open file for writing" + Protocol.CRLF);
            }
        } else {
            enqueue(Protocol.CTRL_ERROR + " unknown command" + Protocol.CRLF);
        }
    }

    /** �������� ��������� � ������� */
    public void enqueue(String msg) {
        outQueue.add(ByteBuffer.wrap(msg.getBytes()));
    }

    /** ������ ������ ������� */
    public void write() throws IOException {
        // ������� ������
        while (!outQueue.isEmpty()) {
            ByteBuffer buf = outQueue.peek();
            channel.write(buf);
            if (!buf.hasRemaining()) outQueue.poll();
            else return;
        }

        // �������� ����� ��������
        if (sendingFile && fileChannel != null) {
            ByteBuffer buf = ByteBuffer.allocate(8192); // ������ ~8KB
            int read = fileChannel.read(buf, filePos);
            if (read == -1) {
                sendingFile = false;
                fileChannel.close();
                enqueue(Protocol.CTRL_DONE + Protocol.CRLF);
                return;
            }
            buf.flip();
            channel.write(buf);
            filePos += read;
        }
    }

    public boolean hasPendingWrites() {
        return !outQueue.isEmpty() || sendingFile;
    }

import java.nio.channels.DatagramChannel;
import java.nio.charset.StandardCharsets;
import java.nio.file.Path;

public class UdpNioContext {
    private final DatagramChannel channel;
    private final SocketAddress clientAddr;
    private final Path serverDir;

    public UdpNioContext(DatagramChannel channel, SocketAddress clientAddr, Path serverDir) {
        this.channel = channel;
        this.clientAddr = clientAddr;
        this.serverDir = serverDir;
    }

    public SocketAddress remote() {
        return clientAddr;
    }

    public Path filesDir() {
        return serverDir;
    }

    public void sendLine(String line) throws IOException {
        ByteBuffer out = StandardCharsets.UTF_8.encode(line + Protocol.CRLF);
        channel.send(out, clientAddr);
    }

    public void sendChunk(byte[] bytes, int off, int len) throws IOException {
        ByteBuffer out = ByteBuffer.wrap(bytes, off, len);
        channel.send(out, clientAddr);
    }

    public int chunkSize() {
        return Protocol.UDP_CHUNK_SIZE;
    }
}
